# MonaCode Command Dispatcher Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the A3 command dispatcher — the component that turns a resolved `commandId` into a model edit or selection change — for 9 frozen core commands, TDD-verified against monaco-editor@0.56.0 oracle fixtures.

**Architecture:** A thin `MonaCommandDispatcher` sibling type (Core, Foundation-only) mirroring monaco's flat `CommandsRegistry`+`ICommandService` and MonaCode's `MonaAXMutationGateway` precedent. Constructor-injected with `model`/`inputBarrier`/`transactionGateway` (= `inputBarrier.gateway`, same instance)/`caretOps`. Handlers take a `MonaCommandContext` value (no ServicesAccessor in MonaCode; avoids retain cycle). Edit commands commit via `MonaModelInputBarrier`; selection commands via `MonaCaretOperationsFeature.commitCaretMove` + `MonaTransactionGateway`. Owned by `MonaCodeEditorView.performAttach` (sibling to `axMutationGateway`).

**Tech Stack:** Swift (MonaCode Core module, Foundation-only); XCTest (MonaCodeTests); Node + jsdom + monaco-editor@0.56.0 (oracle harness, dev-only, generates frozen JSON fixtures).

**Spec:** `docs/superpowers/specs/2026-08-19-monacode-command-dispatcher-design.md` (converged, all-fact, §8 evidence appendix). The plan argues from the spec; executors read both.

## Global Constraints

- Dispatcher lives in `Sources/MonaCode/Input/MonaCommandDispatcher.swift` — Core, `import Foundation` only (no AppKit). §8.1.
- `transactionGateway` dep MUST be `inputBarrier.gateway` (same instance, `public let` `MonaModelInputBarrier.swift:104`) — never a separately-constructed gateway (else `lastCommittedSelections` diverges between edit/selection paths). §8.4.
- Use the 9 frozen command-id strings verbatim (`MonaCommandRegistry.swift:200-576`): `type`/`deleteLeft`/`deleteRight`/`cursorLeft`/`cursorRight`/`cursorUp`/`cursorDown`/`cursorEnd`/`cursorHome`. §8.9.
- `MonaCaretMoveTarget` has NO `.left/.right/.up/.down` — use `.character(±1)`/`.line(±1)`/`.lineStart`/`.lineEnd`. §8.7.
- `cursorEnd` `sticky` arg is IGNORED (no API landing; equals monaco default `false`). §4.5.
- v1 is single-cursor (`commitCaretMove` moves one position). Multi-cursor move / `*Select` / undo / paste / find are out of scope. §1.
- TDD: Red → Green → Commit per task. Frequent commits.
- Public API of MonaCode is FROZEN at P07-T011 — this adds an internal Core type + tests; it does NOT modify any frozen public API.

```swift
// Sources/MonaCode/Input/MonaCommandDispatcher.swift
import Foundation

public struct MonaCommandContext {
    public let model: MonaCodeModel
    public let inputBarrier: MonaModelInputBarrier
    public let transactionGateway: MonaTransactionGateway
    public let caretOps: MonaCaretOperationsFeature
    public let args: Any?
    public init(model: MonaCodeModel, inputBarrier: MonaModelInputBarrier,
                transactionGateway: MonaTransactionGateway,
                caretOps: MonaCaretOperationsFeature, args: Any?) {
        self.model = model; self.inputBarrier = inputBarrier
        self.transactionGateway = transactionGateway; self.caretOps = caretOps; self.args = args
    }
}

public final class MonaCommandDispatcher {
    private let model: MonaCodeModel
    private let inputBarrier: MonaModelInputBarrier
    private let transactionGateway: MonaTransactionGateway
    private let caretOps: MonaCaretOperationsFeature
    private var handlers: [String: (MonaCommandContext, Any?) -> Void] = [:]

    public init(model: MonaCodeModel, inputBarrier: MonaModelInputBarrier,
                transactionGateway: MonaTransactionGateway,
                caretOps: MonaCaretOperationsFeature) {
        self.model = model; self.inputBarrier = inputBarrier
        self.transactionGateway = transactionGateway; self.caretOps = caretOps
    }
    public func register(_ commandId: String, handler: @escaping (MonaCommandContext, Any?) -> Void) {
        handlers[commandId] = handler
    }
    public func contains(_ commandId: String) -> Bool { handlers[commandId] != nil }
    @discardableResult
    public func execute(_ commandId: String, args: Any? = nil) -> Bool {
        guard let handler = handlers[commandId] else { return false }
        handler(MonaCommandContext(model: model, inputBarrier: inputBarrier,
                                  transactionGateway: transactionGateway,
                                  caretOps: caretOps, args: args), args)
        return true
    }
}
```

```js
window.ResizeObserver = RO; window.trustedTypes = undefined;

function run(model, sel, command, args) {
```

```swift
    // in registerCoreCommands:
    register("deleteLeft") { ctx, _ in Self.deleteLeft(ctx) }

    private static func deleteLeft(_ ctx: MonaCommandContext) {
        let sels = currentSelections(ctx)
        let lineCount = ctx.model.getLineCount()
        let maxCol = { ctx.model.getLineMaxColumn($0) }
        let edits = sels.compactMap { sel -> MonaCursorInputEdit? in
            guard let range = deleteLeftRange(sel, lineCount: lineCount, maxColumnOf: maxCol) else { return nil }
            return MonaCursorInputEdit(range: range, text: "")
        }
        guard let primary = edits.first else { return }
        _ = ctx.inputBarrier.commit(MonaMultiCursorInputPlan(primary: primary, secondary: Array(edits.dropFirst())), overlapPolicy: .reject)
    }

    /// Pure: the range deleteLeft removes at the caret, or nil for no-op.
    /// Non-empty selection → delete its range. Caret at col>1 → delete prev char.
    /// Caret at col 1, line>1 → join previous line (delete from prev-line maxColumn to current col 1).
    static func deleteLeftRange(_ sel: MonaSelection, lineCount: Int, maxColumnOf: (Int) -> Int) -> MonaRange? {
        let range = MonaRange(startPosition: sel.anchor, endPosition: sel.activePosition)   // normalized
        if !range.isFolded { return range }                                                 // delete selection content
        let pos = sel.activePosition
        if pos.column > 1 {
            return MonaRange(startLine: pos.line, startColumn: pos.column - 1, endLine: pos.line, endColumn: pos.column)
        }
        if pos.line > 1 {
            let prevMaxCol = maxColumnOf(pos.line - 1)                                       // length+1 (EOL position)
            return MonaRange(startLine: pos.line - 1, startColumn: prevMaxCol, endLine: pos.line, endColumn: 1)
        }
        return nil                                                                            // line 1 col 1: no-op
    }
```

```swift
    register("deleteRight") { ctx, _ in Self.deleteRight(ctx) }

    private static func deleteRight(_ ctx: MonaCommandContext) {
        let sels = currentSelections(ctx)
        let lineCount = ctx.model.getLineCount()
        let maxCol = { ctx.model.getLineMaxColumn($0) }
        let edits = sels.compactMap { sel -> MonaCursorInputEdit? in
            guard let range = deleteRightRange(sel, lineCount: lineCount, maxColumnOf: maxCol) else { return nil }
            return MonaCursorInputEdit(range: range, text: "")
        }
        guard let primary = edits.first else { return }
        _ = ctx.inputBarrier.commit(MonaMultiCursorInputPlan(primary: primary, secondary: Array(edits.dropFirst())), overlapPolicy: .reject)
    }

    /// Pure: range deleteRight removes at the caret, or nil for no-op.
    /// Non-empty selection → delete its range. Caret col<maxCol → delete next char.
    /// Caret at maxCol, line<lineCount → join next line (delete from caret to next-line col 1).
    static func deleteRightRange(_ sel: MonaSelection, lineCount: Int, maxColumnOf: (Int) -> Int) -> MonaRange? {
        let range = MonaRange(startPosition: sel.anchor, endPosition: sel.activePosition)
        if !range.isFolded { return range }
        let pos = sel.activePosition
        let maxCol = maxColumnOf(pos.line)
        if pos.column < maxCol {
            return MonaRange(startLine: pos.line, startColumn: pos.column, endLine: pos.line, endColumn: pos.column + 1)
        }
        if pos.line < lineCount {
            return MonaRange(startLine: pos.line, startColumn: maxCol, endLine: pos.line + 1, endColumn: 1)
        }
        return nil                                                                            // last line end: no-op
    }
```
